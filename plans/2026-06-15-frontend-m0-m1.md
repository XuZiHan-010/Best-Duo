# Card Game 前端 M0+M1 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建前端脚手架（M0）和登录/大厅界面（M1），实现两名玩家登录进房、准备、房主设置并启动游戏的完整流程，与真实后端 Socket.IO 联调。

**Architecture:** Vite + React + TypeScript 单页应用；zustand 管理从 Socket.IO 接收的服务端状态（`PublicRoomState` / `player:hand`）；`RoomView` 按 `phase` 切换 7 个子视图。所有类型和事件名从 `@take-time/shared` 导入，前端不另立第二套。`mySeatId` 通过存储登录昵称后在 `roomState.seats` 中匹配推导，无需服务端 ack。

**Tech Stack:** React 18, Vite 5, TypeScript 5, Zustand 5, Socket.IO Client 4.7, CSS 变量（无 CSS-in-JS），Google Fonts（Fraunces / Spline Sans / Geist Mono）

```typescript
// client/src/socket/client.ts
import { io, type Socket } from "socket.io-client";

export type ConnectionState = "connecting" | "connected" | "reconnecting" | "disconnected";

type ConnectionCallback = (state: ConnectionState) => void;

const listeners = new Set<ConnectionCallback>();

export function onConnectionChange(cb: ConnectionCallback): () => void {
  listeners.add(cb);
  return () => listeners.delete(cb);
}

function notify(state: ConnectionState) {
  listeners.forEach((cb) => cb(state));
}

// 单例 socket
export const socket: Socket = io({
  autoConnect: false,
  reconnection: true,
  reconnectionDelay: 1000,
  reconnectionDelayMax: 5000,
});

socket.on("connect", () => notify("connected"));
socket.on("disconnect", () => notify("reconnecting"));
socket.on("reconnect", () => notify("connected"));
socket.on("reconnect_failed", () => notify("disconnected"));

// 带昵称参数恢复座位
export function connect(nick?: string) {
  if (nick) {
    // ?nick= 参数供服务端 reconnect 逻辑识别
    (socket.io as { opts: { query: Record<string, string> } }).opts.query = { nick };
  }
  notify("connecting");
  socket.connect();
}

export function disconnect() {
  socket.disconnect();
}
```

```typescript
// client/src/store/useRoomStore.ts
import { create } from "zustand";
import {
  ServerEvents,
  type PublicHandCard,
  type PublicRoomState,
  type RoomErrorPayload,
  type TimerState,
} from "@take-time/shared";
import { socket, onConnectionChange, type ConnectionState } from "../socket/client.js";

interface RoomStore {
  // 服务端下发状态
  roomState: PublicRoomState | null;
  myHand: PublicHandCard[] | null;
  timer: TimerState | null;
  lastError: RoomErrorPayload | null;

  // 客户端本地状态
  myNick: string | null;
  connectionState: ConnectionState;

  // actions
  setRoomState: (s: PublicRoomState) => void;
  setMyHand: (h: PublicHandCard[]) => void;
  setTimer: (t: TimerState | null) => void;
  setLastError: (e: RoomErrorPayload | null) => void;
  setMyNick: (nick: string) => void;
  setConnectionState: (s: ConnectionState) => void;
  clearError: () => void;
}

export const useRoomStore = create<RoomStore>((set) => ({
  roomState: null,
  myHand: null,
  timer: null,
  lastError: null,
  myNick: null,
  connectionState: "connecting",

  setRoomState: (s) => set({ roomState: s }),
  setMyHand: (h) => set({ myHand: h }),
  setTimer: (t) => set({ timer: t }),
  setLastError: (e) => set({ lastError: e }),
  setMyNick: (nick) => set({ myNick: nick }),
  setConnectionState: (s) => set({ connectionState: s }),
  clearError: () => set({ lastError: null }),
}));

// 接线 socket 事件（模块级，只注册一次）
socket.on(ServerEvents.RoomState, (s: PublicRoomState) => {
  useRoomStore.getState().setRoomState(s);
});

socket.on(ServerEvents.PlayerHand, (h: PublicHandCard[]) => {
  useRoomStore.getState().setMyHand(h);
});

socket.on(ServerEvents.TimerSync, (t: TimerState) => {
  useRoomStore.getState().setTimer(t);
});

socket.on(ServerEvents.RoomError, (e: RoomErrorPayload) => {
  useRoomStore.getState().setLastError(e);
});

// 连接状态同步
onConnectionChange((state) => {
  useRoomStore.getState().setConnectionState(state);
  if (state === "disconnected") {
    // 超出重连期：清空房间状态，回到登录态
    useRoomStore.getState().setRoomState(null);
    useRoomStore.getState().setMyNick(null);
  }
});
```

```typescript
// client/src/store/selectors.ts
import type { SeatId } from "@take-time/shared";
import type { RoomStore } from "./useRoomStore.js";

// 注：直接导出纯函数，组件用 useRoomStore(mySeatSelector) 等模式使用

export function mySeatSelector(store: RoomStore) {
  const { roomState, myNick } = store;
  if (!roomState || !myNick) return null;
  return roomState.seats.find((s) => s.nick === myNick) ?? null;
}

export function mySeatIdSelector(store: RoomStore): SeatId | null {
  return mySeatSelector(store)?.id ?? null;
}

export function isHostSelector(store: RoomStore): boolean {
  const seatId = mySeatIdSelector(store);
  return seatId !== null && store.roomState?.host === seatId;
}

export function isMyTurnSelector(store: RoomStore): boolean {
  const seatId = mySeatIdSelector(store);
  if (!seatId || !store.roomState) return false;
  const turn = store.roomState.turn;
  return turn === "race" || turn === seatId;
}

export function hintLeftSelector(store: RoomStore): number {
  const hm = store.roomState?.hintMarkers;
  if (!hm) return 0;
  return hm.total - hm.used;
}

export function canStartSelector(store: RoomStore): boolean {
  if (!isHostSelector(store) || !store.roomState) return false;
  const { seats, ready, capacity } = store.roomState;
  const occupied = seats.filter((s) => s.nick !== null);
  if (occupied.length < capacity) return false;
  return occupied.every((s) => ready[s.id] === true);
}

export function allOccupiedSelector(store: RoomStore): boolean {
  const rs = store.roomState;
  if (!rs) return false;
  return rs.seats.filter((s) => s.nick !== null).length >= rs.capacity;
}
```

```typescript
// client/src/lib/timeFmt.ts

/** 将毫秒数格式化为 MM:SS（如 154000 → "02:34"） */
export function formatMmSs(ms: number): string {
  const totalSeconds = Math.max(0, Math.ceil(ms / 1000));
  const m = Math.floor(totalSeconds / 60);
  const s = totalSeconds % 60;
  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
}

/** 将毫秒数格式化为纯秒整数字符串（如 8000 → "8"） */
export function formatSeconds(ms: number): string {
  return String(Math.max(0, Math.ceil(ms / 1000)));
}
```

```tsx
  value: T;

export function Pill<T extends string | number>({

  value,
  onChange,
  disabled = false,
  name,

          aria-pressed={value === opt.value}
          aria-label={`${name} ${opt.label}`}
        >
          {opt.label}
        </button>
      ))}
    </div>
  );
}
```

```tsx
              seat.id === mySeatId ? "topbar__seat-dot--me" : "",

            ].filter(Boolean).join(" ")}
            title={seat.nick ?? "空缺"}
            aria-label={`座位 ${seat.id}: ${seat.nick ?? "空缺"}${ready[seat.id] ? " 已准备" : ""}${seat.id === host ? " 房主" : ""}`}
          />
        ))}
      </div>

      {hm && (phase === "placing" || phase === "discussion") && (
        <span className="topbar__hints" aria-label={`提示标记剩余 ${hintLeft}`}>
          {Array.from({ length: hm.total }).map((_, i) => (
            <span key={i} aria-hidden="true">{i < hm.used ? "◇" : "◆"}</span>
          ))}
        </span>
      )}

      {remaining !== null && (
        <span
          className={`topbar__timer mono${remaining < 30000 ? " topbar__timer--warn" : ""}`}
          aria-live="polite"
          aria-label={`剩余时间 ${formatMmSs(remaining)}`}
        >
          ⏱ {formatMmSs(remaining)}
        </span>
      )}

      {connLabel && (
        <span
          className={`topbar__conn${isDanger ? " topbar__conn--danger" : ""}`}
          aria-live="assertive"
        >
          {connLabel}
        </span>
      )}
    </header>
  );
}
```
